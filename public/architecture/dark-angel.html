extension.ts — Dark Angel 
/**
 * Activates the extension and registers the night-mode handler.
 */
export function activate(context: ExtensionContext) {
  const theme = {
    background: "#0E0F1A",
    accent:     "#FFCC00",
    radius:     14,
  };
  applyTheme(context, theme);
  console.log("Dark Angel active — welcome to the zone.");
}
