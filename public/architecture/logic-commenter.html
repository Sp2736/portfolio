 function reconcileBalances(ledger: Entry[]) {
  const seen = new Map<string, number>();
  for (const e of ledger) {
    const prev = seen.get(e.acctId) ?? 0;
    seen.set(e.acctId, prev + e.delta);
  }
  return [...seen].filter(([, bal]) => bal !== 0);
