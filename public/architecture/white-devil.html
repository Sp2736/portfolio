extension.ts — White Devil 
// Applies the daylight theme and greets the morning coder
export function activate(context: ExtensionContext) {
  const theme = {
    background: "#FAFAFA",
    accent:     "#CC9900",
    clutter:    0,
  };
  applyTheme(context, theme);
  console.log("White Devil active — code with the light on your side.");
}
